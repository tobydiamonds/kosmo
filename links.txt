byte digits[10] = {
  // dpABCDEFG (assuming common cathode and LSBFIRST)
  B01111110, // 0 : ABCDEF
  B00110000, // 1 : BC
  B01101101, // 2 : ABDEG
  B01111001, // 3 : ABCDG
  B00110011, // 4 : BCFG
  B01011011, // 5 : ACDFG
  B01011111, // 6 : ACDEFG
  B01110000, // 7 : ABC
  B01111111, // 8 : ABCDEFG
  B01110011// 9 : ABCFG
};
